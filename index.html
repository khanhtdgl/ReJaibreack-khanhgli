// Hàm xử lý Tải mã nguồn toàn bộ hệ thống (File ZIP) function downloadSource() { var pwdInput = document.getElementById("sourcePwd").value; var statusText = document.getElementById("downloadStatus"); // Mật khẩu của bạn var correctPwd = "944151"; // Reset trạng thái statusText.innerHTML = ""; if (pwdInput === "") { statusText.style.color = "#ff3b30"; statusText.innerHTML = "⚠️ Bạn chưa nhập mật khẩu!"; return; } if (pwdInput === correctPwd) { statusText.style.color = "#34c759"; // Xanh lá statusText.innerHTML = "✅ Mật khẩu chính xác! Đang tải file ZIP mã nguồn..."; // --- CHỈNH SỬA Ở ĐÂY --- // Thay "ReJailbreak_Source.zip" bằng link URL tới file ZIP của bạn trên GitHub/Host nếu cần // Ví dụ: var zipUrl = "https://raw.githubusercontent.com/user/repo/main/ReJailbreak_Source.zip"; var zipUrl = "ReJailbreak_Source.zip"; var a = document.createElement("a"); a.href = zipUrl; a.download = "ReJailbreak_Source.zip"; // Tên file sẽ lưu về máy document.body.appendChild(a); a.click(); document.body.removeChild(a); // Trả lại trạng thái cũ sau 3 giây setTimeout(function() { document.getElementById("sourcePwd").value = ""; statusText.innerHTML = ""; }, 3000); } else { statusText.style.color = "#ff3b30"; // Đỏ statusText.innerHTML = "❌ Mật khẩu không chính xác!"; document.getElementById("sourcePwd").value = ""; } }
